async function sendQuote() { const rows = document.querySelectorAll(".row-item"); let msg = "📋 【工項報價單】\n------------------\n"; let total = document.getElementById("grand-total").innerText; let hasItem = false; rows.forEach(row => { const select = row.querySelector("select"); const qty = row.querySelector("input").value; if (select.value !== "") { const item = itemList[select.value]; msg += `• ${item.name} x${qty} = NT$ ${(item.price * qty).toLocaleString()}\n`; hasItem = true; } }); if (!hasItem) { alert("請至少選擇一個工項！"); return; } msg += `------------------\n💰 總計：NT$ ${total}`; // 優先嘗試直接發送至當前聊天室 if (typeof liff !== 'undefined' && liff.isApiAvailable('sendMessages')) { try { await liff.sendMessages([{ type: 'text', text: msg }]); liff.closeWindow(); return; } catch (err) { console.log("sendMessages 失敗，改用 shareTargetPicker:", err); } } // 備用方案：跳出 LINE 好友/聊天室選擇器供使用者選擇發送 if (typeof liff !== 'undefined' && liff.isApiAvailable('shareTargetPicker')) { try { await liff.shareTargetPicker([{ type: 'text', text: msg }]); liff.closeWindow(); } catch (err) { alert("傳送失敗：" + err); } } else { alert("目前環境無法調用 LINE 傳送功能，報價如下：\n\n" + msg); } }
